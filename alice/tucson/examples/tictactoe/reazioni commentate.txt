opponent(x,o).
opponent(o,x).
nextID(1).
first(x).
init(grid(_,_,_,_,_,_,_,_,_)).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% history
done(ID,N,Role,Move).
% N da 1 in su, numero della mossa fatta
% Role x oppure o
% Move in tictactoe e' on(X,Y), X e Y cella 
% La tupla indica che la mossa Move numero No e' stata compiuta da Role
% Ce ne sono tante quante le mosse fatte, e nessuno le rimuove

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% current
free(ID,Role).
% La tupla indica che il ruolo Role in ID non e' giocato da umani
% Ce ne sono da zero a due: se ce ne sono due il gioco e' abbandonato
number(ID,N).
% Mossa N correntemente da elaborare
grid(ID,Grid).
% Stato della griglia di gioco come termine grid/9, in tictactoe
% x, o, _ come sottotermini
status(ID,Status).
% Status e':
% -	win(Role) o stalemate quando ID e' finito
% - abandoned se ID abbandonato non finito
% - playing(Role) se sta a Role fare la mossa o dare un comando
% - asking(Role) se Role sta aspettando un suggerimento
% - moving(Role,Move) se Role ha proposto la mossa Move

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% PROTOCOLLI
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% Master agent
% emette le tuple di inizio:
% - first(x)
% - opponent(x,o)
% - opponent(o,x)
% - init(grid(_,_,_,_,_,_,_,_,_))
% - nextID(1)
% con una 
out(init([...]))

reaction(out(init(Tuples)), (
	in_r(init(Tuples)),
	out_r(list(Tuples))
)).
reaction(out_r(list(Tuples)), (
	in_r(list(Tuples))
)).
reaction(out_r(list([Tuple|Tuples])), (
	out_r(Tuple),
	out_r(list(Tuples))
)).

% poi attende tuple con una
in(new(-ID))

reaction(in(new(_)), (post,
	current_tuple(new(ID)),
	rd_r(status(ID,player(Role))),
	rd_r(opponent(Role,Opp)),
	out_r(free(ID,Role)),
	out_r(free(ID,Opp)),
	out_r(join(ID,Role))
)).
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% Expert agent
in(game(+ID,+N,-Grid,-Status))
% Grid e Status di ID prima della mossa numero N
% Se la mossa e' gia' stata fatta (esiste done(ID,N,_,_)) Status e' past
% e l'expert passa a N+1 senza outare nulla
% Se Status e' end (se status(ID,win(Role)), o stalemate, o abandoned)
% l'expert quitta
% Se Status e' proposed(Role,Move) (se status(ID,moving(Role,Move)))
% valuta, e convalida o meno
% Se Status e' turn(Role) (se status(ID,playing(Role)) o asking(Role))
% propone una mossa

% delle reazioni corrispondenti una e solo una ha successo ogni volta

reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(done(ID,N,_,_)),
	out_r(game(ID,N,_,past))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(status(ID,win(_))),
	out_r(game(ID,N,_,end))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(status(ID,stalemate)),
	out_r(game(ID,N,_,end))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(status(ID,abandoned)),
	out_r(game(ID,N,_,end))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(grid(ID,Grid)),
	rd_r(status(ID,moving(Role,Move))),
	out_r(game(ID,N,Grid,proposed(Role,Move)))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(grid(ID,Grid)),
	rd_r(status(ID,asking(Role))),
	out_r(game(ID,N,Grid,turn(Role)))
)).
reaction(in(game(ID,N,_,_)), ( pre,
	rd_r(number(ID,N)),
	rd_r(grid(ID,Grid)),
	rd_r(status(ID,playing(Role))),
	out_r(game(ID,N,Grid,turn(Role)))
)).

out(step(ID,Role,Move,NewN,NewGrid,NewStatus))
% La mossa Move di Role in ID porta alla griglia NewGrid e stato NewStatus 
% Se era una convalida, Move e' quella proposed (in moving/2), 
% e tutto e' effettivamente New.
% Se e' un'invalidazione, i New sono uguali ai precedenti, in particolare 
% NewN e' uguale a N in number(ID,N)
% Se e' una mossa, diventa una done se playing(Role) e free(ID,Role),
% una suggestion se asking(Role), ignorata altrimenti, cancellata comunque

reaction(out(step(ID,Role,Move,NewN,_,_)), (
	rd_r(number(ID,N)),
	NewN is N+1,
	in_r(status(ID,asking(Role))), 
	out_r(status(ID,playing(Role))),
	out_r(suggestion(ID,Role,Move))
)).
reaction(out(step(ID,Role,Move,NewN,NewGrid,NewStatus)), (
	rd_r(number(ID,N)),
	NewN is N+1,
	in_r(status(ID,moving(Role,Move))),
	out_r(status(ID,NewStatus)),
	out_r(ok(ID,N,Role,Move,NewN,NewGrid)),
	out_r(move(ID,Role,Move,ok))
)).
reaction(out(step(ID,Role,Move,N,_,_)), (
	rd_r(number(ID,N)),
	in_r(status(ID,moving(Role,Move))),
	out_r(status(ID,playing(Role))),
	out_r(move(ID,Role,Move,no))
)).
reaction(out(step(ID,Role,Move,NewN,NewGrid,NewStatus)), (
	rd_r(free(ID,Role)),
	rd_r(number(ID,N)),
	NewN is N+1,
	in_r(status(ID,playing(Role))),
	out_r(status(ID,NewStatus)),
	out_r(ok(ID,N,Role,Move,NewN,NewGrid))
)).
reaction(out(step(ID,Role,Move,NewN,NewGrid,NewStatus)), (
	in_r(step(ID,Role,Move,NewN,NewGrid,NewStatus))
)).

reaction(out(ok(ID,N,Role,Move,NewN,NewGrid)), (
	in_r(ok(ID,N,Role,Move,NewN,NewGrid)),
	out_r(done(ID,N,Role,Move)),
	in_r(number(ID,_)),
	out_r(number(ID,NewN)),
	in_r(grid(ID,_)),
	out_r(grid(ID,NewGrid)),
)).

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%% GUI agent
in(join(-ID,-Role))
% GUI agent entra in ID con ruolo Role
% Se c'e' ok, altrimenti lo si crea
reaction(in(join(_,_)), (pre,
	rd_r(free(ID,Role)),
	out_r(join(ID,Role))
)).
reaction(in(join(_,_)), (pre,
	no_r(free(ID,Role)),
	out_r(new)
)).
reaction(in(join(_,_)), (post,
	current_tuple(join(ID,Role)),
	in_r(free(ID,Role))
)).

reaction(out_r(new), (
	in_r(nextID(ID)),
	NewID is ID+1,
	out_r(nextID(NewID)),
	rd_r(init(Grid)),
	out_r(grid(ID,Grid)),
	rd_r(first(Role)),
	out_r(status(ID,playing(Role))),
	out_r(new(ID))
)).

out(leave(ID,Role))
% GUI agent esce da ID, e libera il ruolo Role
reaction(out_r(leave(ID,Role)), (
	in_r(leave(ID,Role))
)).
reaction(out(leave(ID,Role)), (
	rd_r(status(ID,playing(_))),
	out_r(check(ID,Role))
)).
reaction(out(leave(ID,Role)), (
	rd_r(status(ID,asking(_))),
	out_r(check(ID,Role))
)).
reaction(out(leave(ID,Role)), (
	rd_r(status(ID,moving(_,_))),
	out_r(check(ID,Role))
)).

reaction(out_r(check(ID,Role)), (
	in_r(check(ID,Role))
)).
reaction(out_r(check(ID,Role)), (
	rd_r(opponent(Role,Opp)),
	in_r(free(ID,Opp)),
	in_r(status(ID,_)),
	out_r(status(ID,abandoned))
)).
reaction(out_r(check(ID,Role)), (
	rd_r(opponent(Role,Opp)),
	no_r(free(ID,Opp)),
	out_r(free(ID,Role))
)).

in(view(+ID,+Role,-Grid,-Status))
% punto di vista del GUI sul game ID
% Status e' yourTurn, opponentTurn, won, lost o stalemate
reaction(in(view(ID,Role,_,_)), ( pre,

	rd_r(status(ID,playing(Role))), 
	out_r(view(ID,Role,Grid,yourTurn))
)).
reaction(in(view(ID,Role,_,_)), ( pre,
	rd_r(grid(ID,Grid)),
	rd_r(opponent(Role,Opp)), 
	rd_r(status(ID,playing(Opp))), 
	out_r(view(ID,Role,Grid,opponentTurn))
)).
reaction(in(view(ID,Role,_,_)), ( pre,
	rd_r(grid(ID,Grid)),
	rd_r(opponent(Role,Opp)), 
	rd_r(status(ID,moving(Opp,_))), 
	out_r(view(ID,Role,Grid,opponentTurn))
)).
reaction(in(view(ID,Role,_,_)), ( pre,
	rd_r(grid(ID,Grid)),
	rd_r(opponent(Role,Opp)), 
	rd_r(status(ID,asking(Opp))), 
	out_r(view(ID,Role,Grid,opponentTurn))
)).
reaction(in(view(ID,Role,_,_)), ( pre,
	rd_r(grid(ID,Grid)),
	rd_r(status(ID,win(Role))), 
	out_r(view(ID,Role,Grid,won))
)).
reaction(in(view(ID,Role,_,_)), ( pre,
	rd_r(grid(ID,Grid)),
	rd_r(opponent(Role,Opp)), 
	rd_r(status(ID,win(Opp))), 
	out_r(view(ID,Role,Grid,lost))
)).
reaction(in(view(ID,Role,_,_)), ( pre,

)).

in(move(+ID,+Role,+Move,-OK))
% Il giocatore Role propone la mossa Move, validata se OK e' ok.
reaction(in(move(ID,Role,Move,_)), ( pre,
	in_r(status(ID,playing(Role))), 
	out_r(status(ID,moving(Role,Move)))
)).

in(suggestion(+ID,+Role,-Move))
% Il giocatore Role chiede una mossa Move
reaction(in(suggestion(ID,Role,Move)), ( pre,
	in_r(status(ID,playing(Role))), 
	out_r(status(ID,asking(Role)))
)).
